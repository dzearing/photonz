+/// How high a layer's oil paint stands, in points, kept between its strokes
+/// (with the washes still wet, `WetWashes.oil`) so each new stroke is lit
+/// against the paint it lands on (`OilStroke.lit`).
+///
+/// Kept coarse, one value for each cell of about two points, and read back
+/// smoothly between them: what a later stroke stands on is the BODY of the
+/// paint under it, how thick it has built up, not the fine ridges of each
+/// bristle, which the new paint fills. So it costs a sixteenth of the layer's
+/// bitmap at 2x and a 12 megapixel layer's is under 3 MB.
+struct OilHeights {
+    /// Pixels a cell is across.
+    private(set) var cell = 0
+    private(set) var columns = 0
+    private(set) var rows = 0
+    private(set) var values: [Float] = []
+
+    /// How high paint can pile: each stroke over another adds less as the
+    /// paint under it thickens, the way a brush through wet paint pushes
+    /// some of it aside.
+    static let most: Float = 1.6
+
+    var isEmpty: Bool { values.isEmpty }
+    var byteCount: Int { values.count * MemoryLayout<Float>.size }
+
+    /// The paint's height at the middle of pixel (x, y), read smoothly
+    /// between the cells.
+    @inline(__always)
+    func value(_ x: Int, _ y: Int) -> Float {
+        guard !values.isEmpty else { return 0 }
+        let size = Float(cell)
+        let gx = (Float(x) + 0.5) / size - 0.5, gy = (Float(y) + 0.5) / size - 0.5
+        let x0 = Int(gx.rounded(.down)), y0 = Int(gy.rounded(.down))
+        let fx = gx - Float(x0), fy = gy - Float(y0)
+        let a = at(x0, y0), b = at(x0 + 1, y0), c = at(x0, y0 + 1), d = at(x0 + 1, y0 + 1)
+        return (a + (b - a) * fx) * (1 - fy) + (c + (d - c) * fx) * fy
+    }
+
+    /// How the height changes across (x, y), in the same measure as a
+    /// stroke's own (`OilStroke.lit`): twice its change a pixel, left to
+    /// right and top to bottom.
+    @inline(__always)
+    func slope(_ x: Int, _ y: Int) -> (Float, Float) {
+        guard !values.isEmpty else { return (0, 0) }
+        return ((value(x + 2, y) - value(x - 2, y)) * 0.5, (value(x, y + 2) - value(x, y - 2)) * 0.5)
+    }
+
+    @inline(__always)
+    private func at(_ column: Int, _ row: Int) -> Float {
+        let i = min(max(column, 0), columns - 1), j = min(max(row, 0), rows - 1)
+        return values[j * columns + i]
+    }
+
+    /// Adds a stroke's own heights (`field`, in the layer's pixels) to the
+    /// surface, a cell at a time: each cell takes the stroke's mean height
+    /// over it, less as the paint there nears `most`.
+    mutating func add(_ field: CoverageField, scale: CGFloat, width: Int, height: Int) {
+        let box = field.box
+        guard !box.isEmpty, width > 0, height > 0 else { return }
+        if values.isEmpty {
+            cell = max(2, Int((2 * scale).rounded()))
+            columns = (width + cell - 1) / cell
+            rows = (height + cell - 1) / cell
+            values = [Float](repeating: 0, count: columns * rows)
+        }
+        let size = cell, area = Float(cell * cell)
+        let firstColumn = max(0, box.minX / size), lastColumn = min(columns - 1, (box.maxX - 1) / size)
+        let firstRow = max(0, box.minY / size), lastRow = min(rows - 1, (box.maxY - 1) / size)
+        guard firstColumn <= lastColumn, firstRow <= lastRow else { return }
+        let most = Self.most
+        values.withUnsafeMutableBufferPointer { out in
+            guard let out = out.baseAddress else { return }
+            for row in firstRow...lastRow {
+                for column in firstColumn...lastColumn {
+                    var sum: Float = 0
+                    for y in row * size..<(row + 1) * size {
+                        for x in column * size..<(column + 1) * size { sum += field.value(x, y) }
+                    }
+                    guard sum > 0 else { continue }
+                    let i = row * columns + column
+                    out[i] += sum / area * max(0, 1 - out[i] / most)
+                }
+            }
+        }
+    }
+
+    /// The heights the `earlier` strokes of a layer leave, worked out by
+    /// painting them (`PaintRasterizer.paint`) onto scratch paper `limit`
+    /// pixels big at `scale`: for a live stroke handed only those strokes.
+    static func after(_ earlier: [BrushStroke], scale: CGFloat, limit: PixelBox) -> OilHeights {
+        guard earlier.contains(where: { $0.settings.medium == .oil && $0.settings.mode == .paint }),
+              limit.width > 0, limit.height > 0,
+              let context = CGContext(
+                data: nil, width: limit.maxX, height: limit.maxY, bitsPerComponent: 8, bytesPerRow: limit.maxX * 4,
+                space: CGColorSpace(name: CGColorSpace.sRGB) ?? CGColorSpaceCreateDeviceRGB(),
+                bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return OilHeights() }
+        context.translateBy(x: 0, y: CGFloat(limit.maxY))
+        context.scaleBy(x: scale, y: -scale)
+        var wet = WetWashes()
+        for stroke in earlier { PaintRasterizer.paint(stroke, in: context, scale: scale, wet: &wet) }
+        return wet.oil
+    }
